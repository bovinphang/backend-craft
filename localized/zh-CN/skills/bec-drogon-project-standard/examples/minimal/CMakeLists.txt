cmake_minimum_required(VERSION 3.16)
project(craft_drogon_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)
add_executable(craft-server main.cpp)
target_link_libraries(craft-server PRIVATE Drogon::Drogon)
add_executable(contract-test contract_test.cpp)
target_link_libraries(contract-test PRIVATE Drogon::Drogon)
enable_testing()
add_test(NAME rejects_filter_continuation COMMAND contract-test)
